                document.querySelectorAll('.tab-btn').forEach(b => {
                    b.className =
                        'tab-btn px-5 py-2 text-sm font-bold rounded-xl transition-all text-gray-500 hover:text-gray-700';
                });
                document.getElementById('panel-' + active).classList.remove('hidden');
                document.getElementById('tab-' + active).className =
                    'tab-btn px-5 py-2 text-sm font-bold rounded-xl transition-all bg-white text-indigo-700 shadow';
            }
        </script>
